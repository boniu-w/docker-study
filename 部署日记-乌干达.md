## 3. 备份数据库

### 1. 140536 定时运行脚本(mysqlbackup.sh), 备份数据库

mysqlbackup.sh

```shell
#!/bin/bash

# ================= 配置区域 =================
MYSQL_CONTAINER="mysql-master"
MYSQL_USER="root"
MYSQL_PASSWORD="CI@yanfa@2023"

DB_LIST=$(docker exec "$MYSQL_CONTAINER" mysql -u"$MYSQL_USER" -p"$MYSQL_PASSWORD" -N -e "
SELECT schema_name 
FROM information_schema.schemata 
WHERE schema_name NOT IN ('information_schema','performance_schema','sys','mysql');
")

# 根备份目录
ROOT_DIR="/opt/data/mysql-8.0.45/backup"
# ===========================================

# 获取当前日期
DATE=$(date +%Y-%m-%d)

# 生成今日备份目录
TODAY_DIR="$ROOT_DIR/$DATE"
mkdir -p "$TODAY_DIR"

echo "========================================"
echo "开始备份任务"
echo "日期: $DATE"
echo "目标目录: $TODAY_DIR"
echo "========================================"

# ---------------- 循环：遍历数据库 ----------------
for DB in $DB_LIST; do
    echo "----------------------------------------"
    echo "正在备份数据库: [$DB]"

    # 为每个数据库创建子文件夹
    DB_DIR="$TODAY_DIR/$DB"
    mkdir -p "$DB_DIR"

    # 定义文件名：数据库名_日期.sql.gz
    FILE_NAME="${DB}_${DATE}.sql.gz"
    TARGET_PATH="$DB_DIR/$FILE_NAME"

    # 【修改处】使用 -u 和 -p 传递账号密码
    # 注意：-p和密码之间没有空格
    docker exec "$MYSQL_CONTAINER" mysqldump -u"$MYSQL_USER" -p"$MYSQL_PASSWORD" \
      --single-transaction --quick "$DB" \
      | gzip > "$TARGET_PATH"
    
    if [ $? -eq 0 ]; then
        echo "   数据库 [$DB] 备份成功: $FILE_NAME"
        # 显示文件大小
        ls -lh "$TARGET_PATH" | awk '{print "   文件大小: "$5}'
    else
        echo "   数据库 [$DB] 备份失败!"
    fi
done

echo "========================================"
echo "所有备份任务结束。"

# ---------------- 清理旧备份 ----------------
echo "正在清理 15 天前的备份文件夹..."
find "$ROOT_DIR" -mindepth 1 -maxdepth 1 -type d -mtime +15 -exec rm -rf {} \;

echo "清理完成。"
```

### 1. 140536 定时运行脚本(redisbackup.sh)

redisbackup.sh

```sh
#!/bin/bash

# --- 配置区域 ---
# Redis 数据根目录
REDIS_ROOT="/opt/data/redis"
# 备份保存天数
DAYS_TO_KEEP=15
# 备份目录路径
BACKUP_DIR="$REDIS_ROOT/redis_backups"

# --- 逻辑开始 ---

# 1. 确保备份目录存在
mkdir -p $BACKUP_DIR

# 2. 定义备份文件名
BACKUP_FILE="$BACKUP_DIR/backup_$(date +%Y%m%d).tar.gz"

echo "🚀 开始备份 Redis 数据..."

# 3. 执行打包
# -C: 切换到指定目录
# data: 只打包 data 文件夹（相对路径）
cd $REDIS_ROOT && tar -czf $BACKUP_FILE data

# 检查打包是否成功 (如果失败则不执行清理，防止数据丢失)
if [ $? -eq 0 ]; then
    echo "✅ 备份成功: $BACKUP_FILE"
    
    # 4. 清理旧备份
    echo "🧹 正在清理 ${DAYS_TO_KEEP} 天前的旧备份..."
    # -name: 只匹配 backup_ 开头的文件
    # -mtime +15: 15天前的文件
    find $BACKUP_DIR -maxdepth 1 -name "backup_*.tar.gz" -mtime +$DAYS_TO_KEEP -delete
else
    echo "❌ 备份失败，请检查磁盘空间或权限！"
    exit 1
fi
```

```sh
#!/bin/bash

# 配置
ALIAS="miniowgd"
DAYS_TO_KEEP=15
BACKUP_ROOT="/opt/data/minio/minio_backups"
TARGET_DIR="$BACKUP_ROOT/$(date +%Y%m%d)"

# 确保目录存在
mkdir -p $TARGET_DIR

# 核心命令：递归镜像所有桶
# 注意：这会同步所有桶的数据到 /opt/minio_backups/桶名 目录下
mc mirror $ALIAS $TARGET_DIR

# 4. 清理旧数据
echo "🧹 正在清理 ${DAYS_TO_KEEP} 天前的备份..."
# -mtime +5: 查找修改时间在 5 天前的目录
# -type d: 只查找目录
# -exec rm -rf: 执行删除操作
find $BACKUP_ROOT -maxdepth 1 -type d -mtime +$DAYS_TO_KEEP -exec rm -rf {} \;

echo "✅ 备份及清理完成"
```
